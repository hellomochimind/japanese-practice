function getJapaneseVoice() { const voices = window.speechSynthesis.getVoices(); const japaneseVoices = voices.filter(v => v.lang.toLowerCase().startsWith("ja") ); // Prefer Japanese voices with a lighter, brighter sound. const preferred = japaneseVoices.find(v => /nanami|haruka|kyoko|ayumi/i.test(v.name) ); return preferred || japaneseVoices.find(v => v.lang.toLowerCase() === "ja-jp" ) || japaneseVoices[0] || null; } function say(t) { if (!("speechSynthesis" in window)) return; const synth = window.speechSynthesis; synth.cancel(); const u = new SpeechSynthesisUtterance(t); u.lang = "ja-JP"; u.volume = 1.0; // Brighter, lighter and more child-friendly delivery. u.pitch = 1.65; u.rate = 1.0; const voice = getJapaneseVoice(); if (voice) { u.voice = voice; } synth.resume(); synth.speak(u); }
